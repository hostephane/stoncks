Schéma PostgreSQL / PostGIS Unifié AFC + USC

Modèle prêt à recevoir les flux CSV de ventes et les webhooks API KYC

-- 1. Référentiel Géographique (Monde > Région > Pays > Restaurant)
CREATE TABLE afc_restaurants (
  id VARCHAR(10) PRIMARY KEY, -- ex: 'PAR', 'BLR', 'NYC'
  country_iso VARCHAR(3) NOT NULL, -- '250' (FR), '356' (IN)...
  region_id VARCHAR(10) NOT NULL,  -- 'EU', 'APAC', 'NA', 'LATAM', 'MEA'
  entity_source VARCHAR(20) DEFAULT 'AFC', -- 'AFC' ou 'USC_MERGER'
  name VARCHAR(120) NOT NULL
);

-- 2. Ventes Quotidiennes (Ingérées depuis CSV Salesforce)
CREATE TABLE daily_sales (
  id BIGSERIAL PRIMARY KEY,
  sale_date DATE NOT NULL,
  restaurant_id VARCHAR(10) REFERENCES afc_restaurants(id),
  product_category VARCHAR(40) NOT NULL, -- 'Tenders', 'Buckets', 'Spicy Wings'...
  units_sold INT NOT NULL,
  revenue_eur NUMERIC(12,2) NOT NULL
);

-- 3. Retours Campagne KYC (Ingérés via REST API + Enrichis IA/ML)
CREATE TABLE kyc_campaign_feedback (
  id BIGSERIAL PRIMARY KEY,
  restaurant_id VARCHAR(10) REFERENCES afc_restaurants(id),
  product_mentioned VARCHAR(40),
  customer_name VARCHAR(120),
  rating SMALLINT CHECK (rating BETWEEN 1 AND 5),
  ai_sentiment VARCHAR(20), -- 'positif', 'neutre', 'critique'
  ai_confidence NUMERIC(4,3),
  comment_text TEXT NOT NULL,
  received_at TIMESTAMPTZ DEFAULT NOW()
);
